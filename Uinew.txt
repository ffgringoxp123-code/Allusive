local function tween(obj, t, props, style, dir)
    style = style or Enum.EasingStyle.Quad
    dir = dir or Enum.EasingDirection.Out
    local ti = TweenInfo.new(t, style, dir)
    local t = game:GetService("TweenService"):Create(obj, ti, props)
    t:Play()
    return t
end

local function make(class, props)
    local i = Instance.new(class)
    for k, v in pairs(props) do
        if k ~= "Parent" and k ~= "Children" then
            i[k] = v
        end
    end
    if props.Children then
        for _, c in ipairs(props.Children) do
            c.Parent = i
        end
    end
    if props.Parent then
        i.Parent = props.Parent
    end
    return i
end

local function ripple(button, color)
    color = color or Color3.fromRGB(255, 255, 255)
    button.MouseButton1Click:Connect(function()
        local r = make("Frame", {
            Parent = button,
            BackgroundColor3 = color,
            BackgroundTransparency = 0.5,
            BorderSizePixel = 0,
            Size = UDim2.new(0, 0, 0, 0),
            Position = UDim2.new(0.5, 0, 0.5, 0),
            AnchorPoint = Vector2.new(0.5, 0.5),
            ZIndex = 100,
            ClipsDescendants = true
        })
        local corner = make("UICorner", {
            Parent = r,
            CornerRadius = UDim.new(1, 0)
        })
        local s = math.max(button.AbsoluteSize.X, button.AbsoluteSize.Y) * 2
        tween(r, 0.5, {Size = UDim2.new(0, s, 0, s), BackgroundTransparency = 1})
        task.delay(0.5, function() if r then r:Destroy() end end)
    end)
end

local function drag(frame, handle)
    handle = handle or frame
    local d = false
    local ds, sp
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            d = true
            ds = input.Position
            sp = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    d = false
                end
            end)
        end
    end)
    handle.InputChanged:Connect(function(input)
        if d and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - ds
            frame.Position = UDim2.new(sp.X.Scale, sp.X.Offset + delta.X, sp.Y.Scale, sp.Y.Offset + delta.Y)
        end
    end)
end

local function safeRead(path)
    local ok, res = pcall(readfile, path)
    if ok then return res end
    return nil
end

local function safeWrite(path, data)
    pcall(writefile, path, data)
end

local function rgbToTable(c)
    return {math.floor(c.R * 255), math.floor(c.G * 255), math.floor(c.B * 255)}
end

local function tableToRgb(t)
    return Color3.fromRGB(t[1] or 0, t[2] or 0, t[3] or 0)
end

local function configSave()
    local data = {}
    for k, v in pairs(GG.Config) do
        if typeof(v) == "Color3" then
            data[k] = {"Color", rgbToTable(v)}
        else
            data[k] = v
        end
    end
    data.Flags = {}
    for k, v in pairs(GG.Flags) do
        data.Flags[k] = v
    end
    safeWrite("SakuraUI_Config.json", game:GetService("HttpService"):JSONEncode(data))
end

local function configLoad()
    local raw = safeRead("SakuraUI_Config.json")
    if not raw then return false end
    local ok, data = pcall(function()
        return game:GetService("HttpService"):JSONDecode(raw)
    end)
    if not ok or not data then return false end
    for k, v in pairs(data) do
        if k == "Flags" then
            for fk, fv in pairs(v) do
                GG.Flags[fk] = fv
            end
        elseif type(v) == "table" and v[1] == "Color" then
            GG.Config[k] = tableToRgb(v[2])
        else
            GG.Config[k] = v
        end
    end
    return true
end

configLoad()

local Connections = {}
local function newConn(name, conn)
    if Connections[name] then
        Connections[name]:Disconnect()
    end
    Connections[name] = conn
end

local function getFlag(flag, default)
    if GG.Flags[flag] == nil then
        if type(default) == "boolean" then
            GG.Flags[flag] = {false}
        elseif type(default) == "number" then
            GG.Flags[flag] = {default}
        elseif type(default) == "string" then
            GG.Flags[flag] = {default}
        else
            GG.Flags[flag] = {default}
        end
    end
    return GG.Flags[flag][1]
end

local function setFlag(flag, val)
    if not GG.Flags[flag] then GG.Flags[flag] = {nil} end
    GG.Flags[flag][1] = val
    configSave()
end

function Library:SelectTab(tab)
    for t, data in pairs(self._tabs) do
        if t ~= tab then
            data.page.Visible = false
            tween(data.button, 0.15, {BackgroundTransparency = 1})
            tween(data.icon, 0.2, {ImageColor3 = GG.Config.Icon})
            tween(data.label, 0.2, {TextColor3 = GG.Config.Text, TextTransparency = 0.2})
        end
    end
    
    tab.Page.Visible = true
    self._ui.ActiveTab = tab
    
    tween(tab._button, 0.15, {BackgroundTransparency = 0.85})
    tween(tab._icon, 0.2, {ImageColor3 = GG.Config.Primary})
    tween(tab._label, 0.2, {TextColor3 = GG.Config.Text, TextTransparency = 0})
    
    local btnPos = tab._button.AbsolutePosition
    local sidebarPos = self._ui.Container.Sidebar.AbsolutePosition
    local sidebarSize = self._ui.Container.Sidebar.AbsoluteSize
    local relativeY = btnPos.Y - sidebarPos.Y - 17
    tween(self._ui.Container.Sidebar.Indicator, 0.3, {Position = UDim2.new(0, 0, 0, relativeY)})
end

    function Mod:CreateSlider(config)
        config = config or {}
        local title = config.Title or "Slider"
        local min = config.Min or 0
        local max = config.Max or 100
        local default = config.Default or 50
        local flag = config.Flag
        local callback = config.Callback or function() end
        local value = flag and getFlag(flag, default) or default
        
        local row = make("Frame", {
            Parent = optionsContainer,
            BackgroundTransparency = 1,
            Size = UDim2.new(1, 0, 0, 40),
            LayoutOrder = #optionsContainer:GetChildren()

            Parent = row,

            Text = title,
            TextColor3 = GG.Config.Text,
            TextXAlignment = Enum.TextXAlignment.Left,
            Font = Enum.Font.Gotham,
            TextSize = 12
        })
        local valLabel = make("TextLabel", {
            Parent = row,
            BackgroundTransparency = 1,
            Size = UDim2.new(0.4, -4, 0, 16),
            Position = UDim2.new(0.6, 0, 0, 0),
            Text = tostring(math.floor(value * 100) / 100),
            TextColor3 = GG.Config.Primary,
            TextXAlignment = Enum.TextXAlignment.Right,
            Font = Enum.Font.GothamSemibold,
            TextSize = 12
        })
        local bar = make("Frame", {
            Parent = row,
            BackgroundColor3 = GG.Config.ToggleOff,
            BorderSizePixel = 0,
            Size = UDim2.new(1, -8, 0, 4),
            Position = UDim2.new(0, 4, 0, 28)
        })
        make("UICorner", {Parent = bar, CornerRadius = UDim.new(1, 0)})
        local fill = make("Frame", {
            Parent = bar,
            BackgroundColor3 = GG.Config.Primary,
            BorderSizePixel = 0,
            Size = UDim2.new((value - min) / (max - min), 0, 1, 0)
        })
        make("UICorner", {Parent = fill, CornerRadius = UDim.new(1, 0)})
        
        local dragging = false
        local function update(input, animate)
            local pos = math.clamp((input.Position.X - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
            value = min + (max - min) * pos
            value = math.floor(value * 100) / 100
            if flag then setFlag(flag, value) end
            if animate ~= false then
                fill.Size = UDim2.new(pos, 0, 1, 0)
            end
            valLabel.Text = tostring(value)
            pcall(callback, value)
        end
        
        bar.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 then
                dragging = true
                update(input, true)
            end
        end)
        
        newConn("slider_drag_" .. tostring(row), game:GetService("UserInputService").InputChanged:Connect(function(input)
            if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
                update(input, true)
            end
        end))
        
        newConn("slider_release_" .. tostring(row), game:GetService("UserInputService").InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 then
                dragging = false
            end
        end))
        
        valLabel.Text = tostring(value)
        fill.Size = UDim2.new((value - min) / (max - min), 0, 1, 0)
        
        return {
            SetValue = function(v)
                value = math.clamp(v, min, max)
                value = math.floor(value * 100) / 100
                if flag then setFlag(flag, value) end
                fill.Size = UDim2.new((value - min) / (max - min), 0, 1, 0)
                valLabel.Text = tostring(value)
            end,
            GetValue = function() return value end
        }
    end

                    local relX = math.clamp((input.Position.X - satBox.AbsolutePosition.X) / satBox.AbsoluteSize.X, 0, 1)
                    local relY = math.clamp((input.Position.Y - satBox.AbsolutePosition.Y) / satBox.AbsoluteSize.Y, 0, 1)
                    s = relX
                    v = 1 - relY
                    satKnob.Position = UDim2.new(relX, -4, relY, -4)
                    updateColor()
                end
            end
        end))
        
        newConn("color_release_" .. tostring(row), game:GetService("UserInputService").InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 then
                draggingHue = false
                draggingSat = false
            end
        end))
        
        preview.MouseButton1Click:Connect(function()
            open = not open
            if open then
                picker.Visible = true
                tween(picker, 0.2, {Size = UDim2.new(1, -8, 0, 120)})
            else
                tween(picker, 0.2, {Size = UDim2.new(1, -8, 0, 0)})
                task.wait(0.2)
                picker.Visible = false
            end
        end)
        
        preview.BackgroundColor3 = value
        satBox.BackgroundColor3 = Color3.fromHSV(h, 1, 1)
        hueKnob.Position = UDim2.new(h, -3, 0.5, -9)
        satKnob.Position = UDim2.new(s, -4, 1 - v, -4)
        
        return {
            SetValue = function(c)
                value = c
                h, s, v = c:ToHSV()
                h = h or 0; s = s or 1; v = v or 1
                preview.BackgroundColor3 = c
                satBox.BackgroundColor3 = Color3.fromHSV(h, 1, 1)
                hueKnob.Position = UDim2.new(h, -3, 0.5, -9)
                satKnob.Position = UDim2.new(s, -4, 1 - v, -4)
                if flag then setFlag(flag, c) end
            end,
            GetValue = function() return value end
        }
    end

    function Mod:CreateKeybind(config)
        config = config or {}
        local title = config.Title or "Keybind"
        local default = config.Default or Enum.UserInputType.MouseButton2
        local flag = config.Flag
        local callback = config.Callback or function() end
        local value = flag and getFlag(flag, default) or default
        if type(value) == "string" then
            value = Enum.UserInputType[value] or value
        end
        local listening = false
        
        local row = make("Frame", {
            Parent = optionsContainer,
            BackgroundTransparency = 1,
            Size = UDim2.new(1, 0, 0, 24),
            LayoutOrder = #optionsContainer:GetChildren()
        })
        make("TextLabel", {
            Parent = row,
            BackgroundTransparency = 1,
            Size = UDim2.new(0.6, 0, 1, 0),
            Position = UDim2.new(0, 4, 0, 0),
            Text = title,
            TextColor3 = GG.Config.Text,
            TextXAlignment = Enum.TextXAlignment.Left,
            Font = Enum.Font.Gotham,
            TextSize = 12
        })
        local btn = make("TextButton", {
            Parent = row,
            BackgroundColor3 = GG.Config.Background,
            BorderSizePixel = 0,
            Size = UDim2.new(0, 50, 0, 20),
            Position = UDim2.new(1, -54, 0.5, -10),
            Text = "",
            TextColor3 = GG.Config.Text,
            Font = Enum.Font.GothamSemibold,
            TextSize = 11,
            AutoButtonColor = false
        })
        make("UICorner", {Parent = btn, CornerRadius = UDim.new(0, 4)})
        make("UIStroke", {
            Parent = btn,
            Color = GG.Config.Border,
            Thickness = 1,
            ApplyStrokeMode = Enum.ApplyStrokeMode.Border
        })
        
        local function getName(v)
            if typeof(v) == "EnumItem" then
                if v.EnumType == Enum.KeyCode then return v.Name end
                if v == Enum.UserInputType.MouseButton2 then return "RMB" end
                if v == Enum.UserInputType.MouseButton1 then return "LMB" end
                if v == Enum.UserInputType.MouseButton3 then return "MMB" end
            end
            return tostring(v)
        end
        
        btn.Text = getName(value)
        
        btn.MouseButton1Click:Connect(function()
            listening = true
            btn.Text = "..."
            btn.TextColor3 = GG.Config.Primary
        end)
        
        newConn("keybind_listen_" .. tostring(btn), game:GetService("UserInputService").InputBegan:Connect(function(input, gp)
            if gp then return end
            if listening then
                if input.UserInputType == Enum.UserInputType.Keyboard then
                    value = input.KeyCode
                elseif input.UserInputType == Enum.UserInputType.MouseButton1 or
                       input.UserInputType == Enum.UserInputType.MouseButton2 or
                       input.UserInputType == Enum.UserInputType.MouseButton3 then
                    value = input.UserInputType
                end
                listening = false
                btn.Text = getName(value)
                btn.TextColor3 = GG.Config.Text
                if flag then setFlag(flag, value) end
                pcall(callback, value)
                return
            end
            if input.UserInputType == Enum.UserInputType.Keyboard then
                if value == input.KeyCode then
                    pcall(callback, value)
                end
            elseif input.UserInputType == value then
                pcall(callback, value)
            end
        end))
        
        return {
            SetValue = function(v)
                value = v
                btn.Text = getName(v)
                if flag then setFlag(flag, v) end
            end,
            GetValue = function() return value end
        }
    end
    
    return Mod
end

function Library:ApplyTheme(themeName)
    local theme = Themes[themeName]
    if not theme then return end
    for k, v in pairs(theme) do
        GG.Config[k] = v
    end
    GG.Config.Theme = themeName
    self:UpdateColors()
    configSave()
end

function Library:UpdateColors()
    local ui = self._ui
    if not ui then return end
    
    local function safeTween(obj, props)
        if obj and obj.Parent then
            tween(obj, 0.3, props)
        end
    end
    
    safeTween(ui.Container, {BackgroundColor3 = GG.Config.Background})
    safeTween(ui.Container.Sidebar, {BackgroundColor3 = GG.Config.Sidebar})
    safeTween(ui.Container.Sidebar.Divider, {BackgroundColor3 = GG.Config.Border})
    safeTween(ui.Container.Sidebar.Indicator, {BackgroundColor3 = GG.Config.Primary})
    safeTween(ui.Container.Sidebar.LogoIcon, {ImageColor3 = GG.Config.Primary})
    safeTween(ui.Container.Sidebar.Title, {TextColor3 = GG.Config.Text})
    safeTween(ui.Container.Sidebar.Minimize, {ImageColor3 = GG.Config.Icon})
    safeTween(ui.Container.Stroke, {Color = GG.Config.Border})
    
    if ui.Blur then
        pcall(function() ui.Blur.Size = 25 end)
    end
    
    for t, data in pairs(self._tabs) do
        safeTween(data.icon, {ImageColor3 = data == self._ui.ActiveTab and GG.Config.Primary or GG.Config.Icon})
        safeTween(data.label, {TextColor3 = GG.Config.Text})
        if t.LeftScroll then
            t.LeftScroll.ScrollBarImageColor3 = GG.Config.Primary
            t.RightScroll.ScrollBarImageColor3 = GG.Config.Primary
        end
    end
end

function SakuraUI:UpdateThemeColors()
    -- Update main frame colors
    TweenService:Create(self.MainFrame, TweenInfo.new(0.3), {
        BackgroundColor3 = self.Settings:Get("Background")
    }):Play()
    TweenService:Create(self.MainStroke, TweenInfo.new(0.3), {
        Color = self.Settings:Get("Border")
    }):Play()
    
    -- Update sidebar
    local sidebarColor = self.Settings:Get("Sidebar") or self.Settings:Get("Background")
    TweenService:Create(self.Sidebar, TweenInfo.new(0.3), {
        BackgroundColor3 = sidebarColor
    }):Play()
    TweenService:Create(self.SidebarCover, TweenInfo.new(0.3), {
        BackgroundColor3 = sidebarColor
    }):Play()
    
    -- Update sidebar divider
    TweenService:Create(self.SidebarDivider, TweenInfo.new(0.3), {
        BackgroundColor3 = self.Settings:Get("Border")
    }):Play()
    
    -- Update indicator
    TweenService:Create(self.TabIndicator, TweenInfo.new(0.3), {
        BackgroundColor3 = self.Settings:Get("Primary")
    }):Play()
    
    -- Update logo icon color
    TweenService:Create(self.LogoIcon, TweenInfo.new(0.3), {
        TextColor3 = self.Settings:Get("Primary")
    }):Play()
    
    -- Update title color
    TweenService:Create(self.UITitle, TweenInfo.new(0.3), {
        TextColor3 = self.Settings:Get("Text")
    }):Play()
    
    -- Update minimize button
    TweenService:Create(self.MinimizeBtn, TweenInfo.new(0.3), {
        ImageColor3 = self.Settings:Get("Icon")
    }):Play()
    
    -- Update tab buttons
    for _, data in pairs(self.TabButtons) do
        TweenService:Create(data.Icon, TweenInfo.new(0.3), {
            TextColor3 = self.Settings:Get("Icon")
        }):Play()
        TweenService:Create(data.Label, TweenInfo.new(0.3), {
            TextColor3 = self.Settings:Get("Text")
        }):Play()
    end
    
    -- Update scrollbar colors
    for _, tab in ipairs(self.Tabs) do
        tab.LeftSection.ScrollBarImageColor3 = self.Settings:Get("Primary")
        tab.RightSection.ScrollBarImageColor3 = self.Settings:Get("Primary")
    end
    
    -- Update module backgrounds within tabs
    for _, tab in ipairs(self.Tabs) do
        for _, side in ipairs({"Left", "Right"}) do
            for _, mod in ipairs(tab.Modules[side]) do
                if mod.Frame then
                    local moduleColor = self.Settings:Get("Module") or 
                        (self.Settings:Get("DarkUI") and Color3.fromRGB(40, 32, 40) or Color3.fromRGB(255, 240, 243))
                    local moduleBorder = self.Settings:Get("ModuleBorder") or
                        (self.Settings:Get("DarkUI") and Color3.fromRGB(60, 45, 55) or Color3.fromRGB(255, 200, 210))
                    
                    TweenService:Create(mod.Frame, TweenInfo.new(0.3), {
                        BackgroundColor3 = moduleColor
                    }):Play()
                    if mod.Border then
                        TweenService:Create(mod.Border, TweenInfo.new(0.3), {
                            Color = moduleBorder
                        }):Play()
                    end
                    
                    -- Update child elements
                    if mod.TitleLabel then
                        TweenService:Create(mod.TitleLabel, TweenInfo.new(0.3), {
                            TextColor3 = self.Settings:Get("Text")
                        }):Play()
                    end
                    if mod.DescLabel then
                        TweenService:Create(mod.DescLabel, TweenInfo.new(0.3), {
                            TextColor3 = self.Settings:Get("TextSecondary") or Color3.fromRGB(180, 120, 120)
                        }):Play()
                    end
                    if mod.Divider then
                        TweenService:Create(mod.Divider, TweenInfo.new(0.3), {
                            BackgroundColor3 = self.Settings:Get("Divider")
                        }):Play()
                    end
                    if mod.ToggleBtn then
                        TweenService:Create(mod.ToggleBtn, TweenInfo.new(0.3), {
                            ImageColor3 = self.Settings:Get("Icon")
                        }):Play()
                    end
                end
            end
        end
    end
end
