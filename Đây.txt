local function Tween(obj, time, props, style, dir)
    style = style or Enum.EasingStyle.Quad
    dir = dir or Enum.EasingDirection.Out
    local t = TweenService:Create(obj, TweenInfo.new(time, style, dir), props)
    t:Play()
    return t
end

local function New(class, props)
    local obj = Instance.new(class)
    for k, v in pairs(props) do
        if k ~= "Children" then
            pcall(function() obj[k] = v end)
        end
    end
    if props.Children then
        for _, c in ipairs(props.Children) do
            c.Parent = obj
        end
    end
    return obj
end

local function Ripple(btn)
    btn.MouseButton1Click:Connect(function()
        local s = math.max(btn.AbsoluteSize.X, btn.AbsoluteSize.Y) * 2
        local r = New("Frame", {
            Parent = btn, BackgroundColor3 = Color3.fromRGB(255, 255, 255),
            BackgroundTransparency = 0.5, BorderSizePixel = 0,
            Size = UDim2.new(0, 0, 0, 0), Position = UDim2.new(0.5, 0, 0.5, 0),
            AnchorPoint = Vector2.new(0.5, 0.5), ZIndex = 100
        })
        New("UICorner", {Parent = r, CornerRadius = UDim.new(1, 0)})
        Tween(r, 0.4, {Size = UDim2.new(0, s, 0, s), BackgroundTransparency = 1})
        task.delay(0.4, function() if r then r:Destroy() end end)
    end)
end

local function Drag(frame)
    local d = false; local ds; local sp
    frame.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then
            d = true; ds = i.Position; sp = frame.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then d = false end
            end)
        end
    end)
    frame.InputChanged:Connect(function(i)
        if d and i.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = i.Position - ds
            frame.Position = UDim2.new(sp.X.Scale, sp.X.Offset + delta.X, sp.Y.Scale, sp.Y.Offset + delta.Y)
        end
    end)
end

local function formatColor(c)
    return {R = math.floor(c.R * 255) / 255, G = math.floor(c.G * 255) / 255, B = math.floor(c.B * 255) / 255}
end

local Connections = {}
local function AddConn(name, conn)
    if Connections[name] then pcall(Connections[name].Disconnect, Connections[name]) end
    Connections[name] = conn
    return conn
end

function Library:SelectTab(tab)
    if self.ActiveTab then
        self.ActiveTab.Page.Visible = false
        Tween(self.ActiveTab.Btn, 0.15, {BackgroundTransparency = 1})
        Tween(self.ActiveTab.BtnIcon, 0.2, {ImageColor3 = Config.Icon})
    end
    tab.Page.Visible = true
    self.ActiveTab = tab
    Tween(tab.Btn, 0.15, {BackgroundTransparency = 0.85})
    Tween(tab.BtnIcon, 0.2, {ImageColor3 = Config.Primary})

    local absY = tab.Btn.AbsolutePosition.Y - self.UI.TabList.AbsolutePosition.Y
    Tween(self.UI.Indicator, 0.3, {Position = UDim2.new(0, 0, 0, absY + 4)})
end

            if i.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
        end))

        fill.Size = UDim2.new((value - min) / (max - min), 0, 1, 0)

                fill.Size = UDim2.new((value - min) / (max - min), 0, 1, 0)
                vLabel.Text = tostring(value)

        }
    end

        New("TextLabel", {
            Parent = row, BackgroundTransparency = 1, Size = UDim2.new(0.6, 0, 1, 0),
            Position = UDim2.new(0, 4, 0, 0), Text = title, TextColor3 = Config.Text,
            TextXAlignment = Enum.TextXAlignment.Left, Font = Enum.Font.Gotham, TextSize = 12
        })
        local btn = New("TextButton", {
            Parent = row, BackgroundColor3 = Config.Background, BorderSizePixel = 0,
            Size = UDim2.new(0, 50, 0, 20), Position = UDim2.new(1, -54, 0.5, -10),
            Text = "", TextColor3 = Config.Text, Font = Enum.Font.GothamSemibold, TextSize = 11, AutoButtonColor = false
        })
        New("UICorner", {Parent = btn, CornerRadius = UDim.new(0, 4)})
        New("UIStroke", {Parent = btn, Color = Config.Border, Thickness = 1, ApplyStrokeMode = Enum.ApplyStrokeMode.Border})

        local function GetName(v)
            if typeof(v) == "EnumItem" then return v.Name end
            if type(v) == "string" then return v end
            return "F"
        end
        btn.Text = GetName(value)

        btn.MouseButton1Click:Connect(function()
            listening = true; btn.Text = "..."; btn.TextColor3 = Config.Primary
        end)

        AddConn("kb_" .. tostring(math.random()), UserInputService.InputBegan:Connect(function(i, gp)
            if gp then return end
            if listening then
                if i.UserInputType == Enum.UserInputType.Keyboard then value = i.KeyCode
                elseif i.UserInputType == Enum.UserInputType.MouseButton1 then value = Enum.UserInputType.MouseButton1
                elseif i.UserInputType == Enum.UserInputType.MouseButton2 then value = Enum.UserInputType.MouseButton2
                elseif i.UserInputType == Enum.UserInputType.MouseButton3 then value = Enum.UserInputType.MouseButton3 end
                listening = false; btn.Text = GetName(value); btn.TextColor3 = Config.Text
                if flag then getgenv().GG.Flags[flag] = value end
                pcall(cb, value)
                return
            end
            if i.UserInputType == Enum.UserInputType.Keyboard and value == i.KeyCode then pcall(cb, value)
            elseif i.UserInputType == value then pcall(cb, value) end
        end))

        return {
            Set = function(v) value = v; btn.Text = GetName(v); if flag then getgenv().GG.Flags[flag] = v end end,
            Get = function() return value end
        }
    end

    return Mod
end

function Tab:Show()
    self.Container.Visible = true
    TweenService:Create(self.Container, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        BackgroundTransparency = 0
    }):Play()
end

function Tab:Hide()
    TweenService:Create(self.Container, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        BackgroundTransparency = 1
    }):Play()
    task.delay(0.2, function()
        self.Container.Visible = false
    end)
end

function Tab:CreateModule(data)
    data = data or {}
    local side = data.Side or "Left"
    local targetContainer = side == "Left" and self.LeftSection or self.RightSection
    
    local module = Module.new(targetContainer, data, self.Library)
    table.insert(self.Modules[side], module)
    self.ModuleCount[side] = self.ModuleCount[side] + 1
    
    self:RepositionModules(targetContainer, side)
    
    return module
end

function Tab:RepositModules(container, side)
    -- Calculate total height and reposition
    local y = 5
    local modules = container:GetChildren()
    for _, child in ipairs(modules) do
        if child:IsA("Frame") and child:FindFirstChild("Header") then
            child.Position = UDim2.new(0, 5, 0, y)
            y = y + child.Size.Y.Offset + 5
        end
    end
    
    if side then
        self.ModuleCount[side] = y
    end
    
    -- Update canvas size
    container.CanvasSize = UDim2.new(0, 0, 0, y + 10)
end

function SakuraUI:ToggleMinimize()
    self.Minimized = not self.Minimized
    
    if self.Minimized then
        -- Minimize: shrink to sidebar
        TweenService:Create(self.MainFrame, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Size = UDim2.new(0, 180, 0, 60),
            Position = UDim2.new(0.5, -90, 0.5, -30)
        }):Play()
        -- Hide content
        TweenService:Create(self.ContentContainer, TweenInfo.new(0.2), {BackgroundTransparency = 1}):Play()
        self.ContentContainer.Visible = false
    else
        -- Maximize: expand to full
        self.ContentContainer.Visible = true
        TweenService:Create(self.ContentContainer, TweenInfo.new(0.2), {BackgroundTransparency = 0}):Play()
        TweenService:Create(self.MainFrame, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Size = UDim2.new(0, 760, 0, 520),
            Position = UDim2.new(0.5, -380, 0.5, -260)
        }):Play()
    end
end

function SakuraUI:RepositionModules(container)
    local y = 5
    local modules = container:GetChildren()
    for _, child in ipairs(modules) do
        if child:IsA("Frame") and child:FindFirstChild("Header") then
            TweenService:Create(child, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Position = UDim2.new(0, 5, 0, y)
            }):Play()
            y = y + child.Size.Y.Offset + 5
        end
    end
    container.CanvasSize = UDim2.new(0, 0, 0, y + 10)
end

function SakuraUI:CreateTab(name, icon)
    icon = icon or ""
    
    -- Create tab data
    local tab = Tab.new({Name = name, Icon = icon}, self)
    table.insert(self.Tabs, tab)
    
    -- Create tab button
    local btn = Create("TextButton", {
        Parent = self.TabContainer,
        BackgroundTransparency = 1,
        Size = UDim2.new(1, 0, 0, 36),
        Position = UDim2.new(0, 0, 0, (#self.Tabs - 1) * 36),
        Text = "",
        ZIndex = 5,
        AutoButtonColor = false
    })
    
    -- Tab icon
    local btnIcon = Create("TextLabel", {
        Parent = btn,
        BackgroundTransparency = 1,
        Size = UDim2.new(0, 24, 1, 0),
        Position = UDim2.new(0, 12, 0, 0),
        Text = icon,
        TextColor3 = self.Settings:Get("Icon"),
        Font = Enum.Font.Gotham,
        TextSize = 16,
        ZIndex = 6
    })
    
    -- Tab label
    local btnLabel = Create("TextLabel", {
        Parent = btn,
        BackgroundTransparency = 1,
        Size = UDim2.new(1, -44, 1, 0),
        Position = UDim2.new(0, 40, 0, 0),
        Text = name,
        TextColor3 = self.Settings:Get("Text"),
        TextXAlignment = Enum.TextXAlignment.Left,
        Font = Enum.Font.Gotham,
        TextSize = 14,
        ZIndex = 6
    })
    
    -- Store reference
    self.TabButtons[tab] = {
        Button = btn,
        Icon = btnIcon,
        Label = btnLabel
    }
    
    -- Update canvas size
    local tabCount = #self.Tabs
    self.TabContainer.CanvasSize = UDim2.new(0, 0, 0, 36 * (tabCount + 1)) -- +1 for settings
    
    -- Click handler
    btn.MouseButton1Click:Connect(function()
        self:SelectTab(tab)
    end)
    
    -- Hover
    btn.MouseEnter:Connect(function()
        if self.ActiveTab ~= tab then
            TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundTransparency = 0.85}):Play()
        end
    end)
    btn.MouseLeave:Connect(function()
        if self.ActiveTab ~= tab then
            TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundTransparency = 1}):Play()
        end
    end)
    
    -- Auto-select first tab
    if #self.Tabs == 1 then
        self:SelectTab(tab)
    end
    
    return tab
end

function SakuraUI:SelectTab(tab)
    -- Hide current tab
    if self.ActiveTab then
        self.ActiveTab:Hide()
    end
    
    -- Show new tab
    self.ActiveTab = tab
    tab:Show()
    
    -- Move indicator
    local btnData = self.TabButtons[tab]
    if btnData then
        local targetY = btnData.Button.Position.Y.Offset
        TweenService:Create(self.TabIndicator, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Position = UDim2.new(0, 2, 0, targetY + 2)
        }):Play()
        
        -- Highlight active tab
        for _, data in pairs(self.TabButtons) do
            data.Button.BackgroundTransparency = (data == btnData) and 0.85 or 1
        end
    end
end

function SakuraUI:UpdateThemeColors()
    -- Update main frame colors
    TweenService:Create(self.MainFrame, TweenInfo.new(0.3), {
        BackgroundColor3 = self.Settings:Get("Background")
    }):Play()
    TweenService:Create(self.MainStroke, TweenInfo.new(0.3), {
        Color = self.Settings:Get("Border")
    }):Play()
    
    -- Update sidebar
    local sidebarColor = self.Settings:Get("Sidebar") or self.Settings:Get("Background")
    TweenService:Create(self.Sidebar, TweenInfo.new(0.3), {
        BackgroundColor3 = sidebarColor
    }):Play()
    TweenService:Create(self.SidebarCover, TweenInfo.new(0.3), {
        BackgroundColor3 = sidebarColor
    }):Play()
    
    -- Update sidebar divider
    TweenService:Create(self.SidebarDivider, TweenInfo.new(0.3), {
        BackgroundColor3 = self.Settings:Get("Border")
    }):Play()
    
    -- Update indicator
    TweenService:Create(self.TabIndicator, TweenInfo.new(0.3), {
        BackgroundColor3 = self.Settings:Get("Primary")
    }):Play()
    
    -- Update logo icon color
    TweenService:Create(self.LogoIcon, TweenInfo.new(0.3), {
        TextColor3 = self.Settings:Get("Primary")
    }):Play()
    
    -- Update title color
    TweenService:Create(self.UITitle, TweenInfo.new(0.3), {
        TextColor3 = self.Settings:Get("Text")
    }):Play()
    
    -- Update minimize button
    TweenService:Create(self.MinimizeBtn, TweenInfo.new(0.3), {
        ImageColor3 = self.Settings:Get("Icon")
    }):Play()
    
    -- Update tab buttons
    for _, data in pairs(self.TabButtons) do
        TweenService:Create(data.Icon, TweenInfo.new(0.3), {
            TextColor3 = self.Settings:Get("Icon")
        }):Play()
        TweenService:Create(data.Label, TweenInfo.new(0.3), {
            TextColor3 = self.Settings:Get("Text")
        }):Play()
    end
    
    -- Update scrollbar colors
    for _, tab in ipairs(self.Tabs) do
        tab.LeftSection.ScrollBarImageColor3 = self.Settings:Get("Primary")
        tab.RightSection.ScrollBarImageColor3 = self.Settings:Get("Primary")
    end
    
    -- Update module backgrounds within tabs
    for _, tab in ipairs(self.Tabs) do
        for _, side in ipairs({"Left", "Right"}) do
            for _, mod in ipairs(tab.Modules[side]) do
                if mod.Frame then
                    local moduleColor = self.Settings:Get("Module") or 
                        (self.Settings:Get("DarkUI") and Color3.fromRGB(40, 32, 40) or Color3.fromRGB(255, 240, 243))
                    local moduleBorder = self.Settings:Get("ModuleBorder") or
                        (self.Settings:Get("DarkUI") and Color3.fromRGB(60, 45, 55) or Color3.fromRGB(255, 200, 210))
                    
                    TweenService:Create(mod.Frame, TweenInfo.new(0.3), {
                        BackgroundColor3 = moduleColor
                    }):Play()
                    if mod.Border then
                        TweenService:Create(mod.Border, TweenInfo.new(0.3), {
                            Color = moduleBorder
                        }):Play()
                    end
                    
                    -- Update child elements
                    if mod.TitleLabel then
                        TweenService:Create(mod.TitleLabel, TweenInfo.new(0.3), {
                            TextColor3 = self.Settings:Get("Text")
                        }):Play()
                    end
                    if mod.DescLabel then
                        TweenService:Create(mod.DescLabel, TweenInfo.new(0.3), {
                            TextColor3 = self.Settings:Get("TextSecondary") or Color3.fromRGB(180, 120, 120)
                        }):Play()
                    end
                    if mod.Divider then
                        TweenService:Create(mod.Divider, TweenInfo.new(0.3), {
                            BackgroundColor3 = self.Settings:Get("Divider")
                        }):Play()
                    end
                    if mod.ToggleBtn then
                        TweenService:Create(mod.ToggleBtn, TweenInfo.new(0.3), {
                            ImageColor3 = self.Settings:Get("Icon")
                        }):Play()
                    end
                end
            end
        end
    end
end

--[[
    Return the library
]]
return Library
