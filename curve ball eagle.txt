local function MakeDraggable(frame)
    local dragging, dragInput, dragStart, startPos
    frame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = frame.Position
            input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
        end
    end)
    frame.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
    end)
    UIS.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset+delta.X, startPos.Y.Scale, startPos.Y.Offset+delta.Y)
        end
    end)
end

local rainbowColors = {
    Color3.fromRGB(255,0,0), Color3.fromRGB(255,165,0),
    Color3.fromRGB(255,255,0), Color3.fromRGB(0,255,0),
    Color3.fromRGB(0,0,255), Color3.fromRGB(255,0,255),
    Color3.fromRGB(0,255,255), Color3.fromRGB(255,105,180),
}
local colorOptions = {
    Black  = Color3.fromRGB(0,0,0),
    Dark   = Color3.fromRGB(30,30,30),
    Blue   = Color3.fromRGB(0,120,255),
    Red    = Color3.fromRGB(255,30,30),
    White  = Color3.fromRGB(255,255,255),
    Pink   = Color3.fromRGB(255,105,180),
    Cyan   = Color3.fromRGB(0,255,255),
}
local colorIndex = 1

local function addTrail(ball, color)
    if not ball:FindFirstChild("EagleTrail") then
        local att0 = Instance.new("Attachment", ball); att0.Name = "TrailAtt0"; att0.Position = Vector3.new(0,1,0)
        local att1 = Instance.new("Attachment", ball); att1.Name = "TrailAtt1"; att1.Position = Vector3.new(0,-1,0)
        local trail = Instance.new("Trail", ball); trail.Name = "EagleTrail"
        trail.Attachment0 = att0; trail.Attachment1 = att1
        trail.Lifetime = 0.3; trail.MinLength = 0
        trail.WidthScale = NumberSequence.new({NumberSequenceKeypoint.new(0,1.5),NumberSequenceKeypoint.new(1,0)})
        trail.Transparency = NumberSequence.new({NumberSequenceKeypoint.new(0,0),NumberSequenceKeypoint.new(0.5,0.3),NumberSequenceKeypoint.new(1,1)})
        trail.LightEmission = 0.5; trail.LightInfluence = 0.5
    end
    local t = ball:FindFirstChild("EagleTrail")
    if t then
        if typeof(color) == "Color3" then
            -- Qara/tünd rənglər üçün fade
            local endColor = color == Color3.fromRGB(0,0,0) and Color3.fromRGB(30,30,30) or Color3.fromRGB(0,0,0)
            t.Color = ColorSequence.new({ColorSequenceKeypoint.new(0,color), ColorSequenceKeypoint.new(1,endColor)})
        end
    end
end

local function removeTrail(ball)
    if ball:FindFirstChild("EagleTrail") then ball.EagleTrail:Destroy() end
    if ball:FindFirstChild("TrailAtt0") then ball.TrailAtt0:Destroy() end
    if ball:FindFirstChild("TrailAtt1") then ball.TrailAtt1:Destroy() end
end

local function setupOrbit()
    for _, v in pairs(orbitParts) do pcall(function() v:Destroy() end) end
    orbitParts = {}
    for i = 1, 6 do
        local part = Instance.new("Part", workspace)
        part.Name = "EagleOrbit"; part.Size = Vector3.new(0.7,0.7,0.7)
        part.Anchored = true; part.CanCollide = false
        part.Material = Enum.Material.Neon; part.CastShadow = false
        part.Shape = Enum.PartType.Ball
        table.insert(orbitParts, part)
    end
end

local fpsTimer = 0
local frameCount = 0
local currentFps = 0

RunService.RenderStepped:Connect(function(dt)
    colorIndex = colorIndex % #rainbowColors + 1
    frameCount = frameCount + 1
    fpsTimer = fpsTimer + dt
    if fpsTimer >= 0.5 then
        currentFps = math.floor(frameCount / fpsTimer)
        frameCount = 0; fpsTimer = 0
    end

    if showPingFps then
        FpsValueLabel.Text = "FPS\n"..currentFps
        FpsValueLabel.TextColor3 = currentFps >= 55 and Color3.fromRGB(0,255,150) or currentFps >= 30 and Color3.fromRGB(255,200,0) or Color3.fromRGB(255,50,50)
        pcall(function()
            local ping = math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue())
            PingValueLabel.Text = "PING\n"..ping.."ms"
            PingValueLabel.TextColor3 = ping <= 80 and Color3.fromRGB(0,255,150) or ping <= 150 and Color3.fromRGB(255,200,0) or Color3.fromRGB(255,50,50)
            local barWidth = math.clamp(currentFps / 60, 0, 1)
            FpsStatusBar.Size = UDim2.new(barWidth, 0, 0, 2)
            FpsStatusBar.BackgroundColor3 = currentFps >= 55 and Color3.fromRGB(0,255,150) or currentFps >= 30 and Color3.fromRGB(255,200,0) or Color3.fromRGB(255,50,50)
        end)
    end

    -- SPAM (çox güclü - 500 CPS)
    if SpamActive and ManualSpam then
        local now = tick()
        if now - lastSpamTick >= SPAM_INTERVAL then
            lastSpamTick = now; DoSpam()
        end
    end

    -- TRAIL
    for _, v in pairs(workspace:GetChildren()) do
        if v:IsA("BasePart") and (v.Name:lower():find("ball") or v:GetAttribute("Ball")) then
            if trailEnabled then
                local color
                if selectedTrailColor == "Rainbow" then
                    color = rainbowColors[colorIndex]
                else
                    color = colorOptions[selectedTrailColor] or Color3.fromRGB(0,0,0)
                end
                addTrail(v, color)
                if selectedTrailColor == "Rainbow" then
                    local t = v:FindFirstChild("EagleTrail")
                    if t then t.Color = ColorSequence.new({ColorSequenceKeypoint.new(0,rainbowColors[colorIndex]), ColorSequenceKeypoint.new(1,rainbowColors[(colorIndex%#rainbowColors)+1])}) end
                end
            else removeTrail(v) end
        end
    end

    -- ESP
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local head = p.Character:FindFirstChild("Head")
            if head then
                local bb = head:FindFirstChild("EagleESP")
                if espEnabled then
                    if not bb then
                        local billboard = Instance.new("BillboardGui", head)
                        billboard.Name = "EagleESP"; billboard.Size = UDim2.new(0,200,0,50)
                        billboard.AlwaysOnTop = true; billboard.ExtentsOffset = Vector3.new(0,3,0)
                        local label = Instance.new("TextLabel", billboard)
                        label.Size = UDim2.new(1,0,1,0); label.BackgroundTransparency = 1
                        label.Text = p.DisplayName; label.TextColor3 = Color3.new(1,1,1)
                        label.Font = Enum.Font.GothamBold; label.TextSize = 18
                        label.TextStrokeTransparency = 0; label.TextStrokeColor3 = Color3.new(0,0,0)
                    end
                else if bb then bb:Destroy() end end
            end
        end
    end

    -- AUTO JUMP
    if autoJumpEnabled and not hasJumped then
        local char = LocalPlayer.Character
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum and hum.FloorMaterial ~= Enum.Material.Air then
                hum.Jump = true; hasJumped = true
                hum:GetPropertyChangedSignal("FloorMaterial"):Connect(function()
                    if hum.FloorMaterial ~= Enum.Material.Air then hasJumped = false end
                end)
            end
        end
    end

    -- ORBIT
    if orbitEnabled then
        orbitAngle = orbitAngle + 0.04
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            if #orbitParts == 0 then setupOrbit() end
            local radius = 5; local count = #orbitParts
            for i, part in pairs(orbitParts) do
                if part and part.Parent then
                    local angle = orbitAngle + (i-1)*(math.pi*2/count)
                    local vertAngle = orbitAngle*0.7 + (i-1)*(math.pi/count)
                    part.CFrame = CFrame.new(hrp.Position + Vector3.new(
                        radius*math.cos(angle)*math.cos(vertAngle),
                        radius*math.sin(vertAngle),
                        radius*math.sin(angle)*math.cos(vertAngle)
                    ))
                    part.Color = rainbowColors[((colorIndex+i)%#rainbowColors)+1]
                end
            end
        end
    else
        if #orbitParts > 0 then
            for _, v in pairs(orbitParts) do pcall(function() v:Destroy() end) end
            orbitParts = {}
        end
    end
end)

-- AUTO PARRY (300+ sürətdə də işləyir)
RunService.PreSimulation:Connect(function()
    local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not hrp or not AutoParry then return end
    local ball = nil
    for _, v in pairs(workspace:GetChildren()) do
        if v:IsA("BasePart") and v.AssemblyLinearVelocity.Magnitude > 5 and (v.Name:lower():find("ball") or v:GetAttribute("Ball")) then
            ball = v; break
        end
    end
    if ball then
        local vel = ball.AssemblyLinearVelocity.Magnitude
        VLog.Text = math.floor(vel)..".0"
        if vel > peakVel then peakVel = vel; PLog.Text = math.floor(peakVel)..".0" end
        local target = ball:GetAttribute("target") or ball:GetAttribute("Target")
        if target ~= LocalPlayer.Name then return end
        -- Cooldown çox kiçik — 300+ sürətdə belə işləyir
        local cooldown = 0.05
        if tick() - lastHitTick < cooldown then return end
        -- Range çox genişdir — 300+ sürətdə top uzaqdan gəlir
        local range = math.clamp(20 + (vel * 0.35), 30, 300)
        if (hrp.Position - ball.Position).Magnitude < range then
            lastHitTick = tick()
            if hitSoundEnabled then playHitSound() end
            local cam = workspace.CurrentCamera
            local oldCF = cam.CFrame; local x, y, z = 0, 0, 0
            if CurrentMode == "Random" then x,y,z = math.random(-360,360),math.random(-360,360),math.random(-360,360)
            elseif CurrentMode == "Up" then x = -90
            elseif CurrentMode == "Side" then y = math.random(1,2)==1 and 90 or -90
            elseif CurrentMode == "Backward" then x = 180 end
            if CurrentMode ~= "Camera" and (x~=0 or y~=0 or z~=0) then
                cam.CFrame = oldCF * CFrame.Angles(math.rad(x), math.rad(y), math.rad(z))
                DoParry()
                -- Kamera dərhal geri qayıdır
                task.delay(0.08, function() cam.CFrame = oldCF end)
            else
                DoParry()
            end
        end
    else
        VLog.Text = "0.0"
    end
end)
